    ctx := context.Background()

        openrouter.WithSecurity(os.Getenv("OPENROUTER_API_KEY")),

    if err != nil {
        log.Fatal(err)

            // handle items

            res, err = res.Next()

            if err != nil {
                // handle error
            }

            if res == nil {
                break
            }
        }
    }
}
